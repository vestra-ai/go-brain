## Global Constraints

Master §7 applies in full. Added for the demo:
- **Honesty on stage.** Anything simulated is said out loud and labelled on screen ("Simulated for the demo"). The board shows only numbers measured from real turns, with the window it covers.
- **The real trip is real.** Its history is what the team actually said; nothing is seeded into it.
- **Calls go only to the allowlist** (`CALLS_ALLOWLIST`): the teammate playing the villa, or a venue that agreed in writing.
- **Freeze at Fri 18:00.** After the freeze only fixes for a failed rehearsal beat merge, each with the conductor's OK.

## Review Focus

1. **A live dependency fails mid-demo** (ixigo slow, the task stalls, the call is not answered). The operator flips the switch on the admin page within 10 seconds; Go says something true; the presenter says what happened. (Task 1 switch tests; Task 5 run sheet fallback moves.)
2. **State drifts between rehearsals** (an old vote open, an old pay card waiting). Each rehearsal starts on a fresh trip made by script; the real trip is used once, on stage. (Task 3 runner creates its own trip; Task 5 reset steps.)
3. **A cold start on stage** (empty cache, cold computer, cold Jev connection). A warm-up 10 minutes before the demo: the health read for the real trip, and one real message from a phone; the board window says what it covers. (Task 3 `--warm`, which only reads `/admin/health.json?trip=`; Task 5 checklist.)
4. **The network on stage drops.** Phones on a hotspot, the App reconnects, Go's typing and posts resume; the backup video is one click away. (Task 5 checklist; Task 7.)
5. **A rehearsal message reaches a real person** (a call or a push to someone outside the team). Calls only to the allowlist; rehearsal trips have only team members; full rehearsals run only on `brain-rehearsal`, never the stage Go (master §9), and P1's trip affinity is the safety net. (Task 1 shows the allowlist; Task 3 refuses to run if `CALLS_ALLOWLIST` is empty while `CALLS_MODE=allowlist`, and refuses a full run on the stage Go; P1 Tasks 4 and 5 test the affinity.)

### Task 1: The admin page and health

The operator's desk at the side of the stage, on the brain service itself. One page shows whether each dependency answered within 1.5 s, every demo switch with its value and a way to change it, who Go may phone, a "Fire a watch" form and two clean-up buttons, and links to the proof board and the docs. `GET /admin/health.json` gives the rehearsal runner (Task 3) the same facts. Every action goes through a lane's own interface: switches through `brain.switches.set`, the watch through P6's `POST /admin/watch/fire`, jobs through `brain.tasks.stop`, and the pay card through the payer's own Decline tap (P5's `declineTap`, reached through P1's tap router). Without `ADMIN_TOKEN` none of it exists (404); with it, every route needs the token as `Authorization: Bearer` or, for the browser page, `?token=`.

import { describe, expect, it } from 'vitest';
import { v1UserIdFor } from '../../tasks/computer.js';
import { checkAll, jevProbe, modelProbe, PROBE_TIMEOUT_MS, runProbe, serverProbe, v1Probe, vaultProbe, type Probe } from '../health.js';

const after = <T>(ms: number, value: T): Promise<T> => new Promise((resolve) => setTimeout(() => resolve(value), ms));
const probe = (run: Probe['run']): Probe => ({ name: 'x', label: 'Something', run });

function fakeFetch(answer: (url: string, init?: RequestInit) => Response) {
  const calls: Array<{ url: string; init: RequestInit | undefined }> = [];
  const f = (async (input: string | URL | Request, init?: RequestInit) => {
    const url = String(input instanceof Request ? input.url : input);
    calls.push({ url, init });
    return answer(url, init);
  }) as typeof fetch;
  return { f, calls };
}

describe('runProbe', () => {
  it('a quick, good answer is green, with the time it took', async () => {
    const r = await runProbe(probe(async () => ({ ok: true, detail: 'open' })));
    expect(r).toMatchObject({ name: 'x', label: 'Something', dot: 'green', detail: 'open' });
    expect(r.ms).toBeLessThan(100);
  });

  it('a slow answer is amber and says so', async () => {
    const r = await runProbe(probe(() => after(800, { ok: true, detail: 'reachable' })));
    expect(r).toMatchObject({ dot: 'amber', detail: 'reachable, slow' });
    expect(r.ms).toBeGreaterThanOrEqual(750);
  });

  it('reachable but not ready is amber', async () => {
    const r = await runProbe(probe(async () => ({ ok: true, degraded: true, detail: 'computer waking' })));
    expect(r).toMatchObject({ dot: 'amber', detail: 'computer waking' });
  });

  it('a failure is red with its reason, and so is a throw', async () => {
    expect(await runProbe(probe(async () => ({ ok: false, detail: 'HTTP 503' })))).toMatchObject({ dot: 'red', detail: 'HTTP 503' });
    const thrown = await runProbe(probe(async () => {
      throw new Error('connect ECONNREFUSED 10.0.0.1:443');
    }));
    expect(thrown).toMatchObject({ dot: 'red', detail: 'connect ECONNREFUSED 10.0.0.1:443' });
  });

  it('gives up at 1.5 s: red, and the check is told to stop', async () => {
    let aborted = false;
    const r = await runProbe(probe((signal) => new Promise((_, reject) => {
      signal.addEventListener('abort', () => {
        aborted = true;
        reject(new Error('aborted'));
      });
    })));
    expect(PROBE_TIMEOUT_MS).toBe(1_500);
    expect(r).toMatchObject({ dot: 'red', detail: 'no answer in 1.5 s' });
    expect(r.ms).toBeGreaterThanOrEqual(1_450);
    expect(r.ms).toBeLessThan(2_000);
    expect(aborted).toBe(true);
  });

  it('runs every check at once, so the page waits for the slowest only', async () => {
    const started = Date.now();
    const rs = await checkAll([1, 2, 3].map(() => probe(() => after(300, { ok: true, detail: 'ok' }))));
    expect(rs.map((r) => r.dot)).toEqual(['green', 'green', 'green']);
    expect(Date.now() - started).toBeLessThan(800);
  });
});

// Health checks for the operator's desk. Each check gets 1.5 s: a slow answer is amber, a failure or no answer is red.
// Every check takes its fetch from outside, so the tests run offline.
import type { Brain } from '../app.js';

export type Dot = 'green' | 'amber' | 'red';

export interface ProbeOutcome {
  ok: boolean;
  detail: string;
  /** Reachable but not ready yet, such as a trip's computer that is still waking. */
  degraded?: boolean;
}

export interface Probe {
  name: string;
  label: string;
  run(signal: AbortSignal): Promise<ProbeOutcome>;
}

export interface ProbeResult {
  name: string;
  label: string;
  dot: Dot;
  ms: number;
  detail: string;
}

export const PROBE_TIMEOUT_MS = 1_500;
/** Slower than this is amber: it works, but people would notice on stage. */
export const SLOW_MS = 750;

export interface ProbeTiming {
  timeoutMs?: number;
  slowMs?: number;
}

type Fetch = typeof fetch;
const clip = (s: string, n = 160): string => (s.length > n ? `${s.slice(0, n - 1)}…` : s);
const trimSlash = (url: string): string => url.replace(/\/+$/, '');

export async function runProbe(p: Probe, t: ProbeTiming = {}): Promise<ProbeResult> {
  const timeoutMs = t.timeoutMs ?? PROBE_TIMEOUT_MS;
  const slowMs = t.slowMs ?? SLOW_MS;
  const ctrl = new AbortController();
  const started = performance.now();
  let timer: NodeJS.Timeout | undefined;
  const late = new Promise<'late'>((resolve) => {
    timer = setTimeout(() => {
      ctrl.abort();
      resolve('late');
    }, timeoutMs);
  });
  const result = (dot: Dot, detail: string): ProbeResult => ({
    name: p.name, label: p.label, dot, ms: Math.round(performance.now() - started), detail: clip(detail),
  });
  try {
    const out = await Promise.race([p.run(ctrl.signal), late]);
    if (out === 'late') return result('red', `no answer in ${timeoutMs / 1000} s`);
    if (!out.ok) return result('red', out.detail);
    if (out.degraded) return result('amber', out.detail);
    return performance.now() - started > slowMs ? result('amber', `${out.detail}, slow`) : result('green', out.detail);
  } catch (e) {
    return result('red', e instanceof Error ? e.message : String(e));
  } finally {
    clearTimeout(timer);
  }
}

/** Every check at once: the page waits for the slowest, never for the sum. */
export function checkAll(probes: Probe[], t: ProbeTiming = {}): Promise<ProbeResult[]> {
  return Promise.all(probes.map((p) => runProbe(p, t)));
}

export function v1Probe(env: Pick<Env, 'VESTRA_API_URL' | 'VESTRA_API_KEYS'>, f: Fetch, tripId: string | null): Probe {
  return {
    name: 'v1',
    label: tripId ? "/v1 and this trip's computer" : '/v1',
    async run(signal) {
      const key = env.VESTRA_API_KEYS?.split(',')[0]?.trim();
      if (!env.VESTRA_API_URL || !key) return { ok: false, detail: 'VESTRA_API_URL or VESTRA_API_KEYS is not set' };
      const base = trimSlash(env.VESTRA_API_URL);
      const headers = { authorization: `Bearer ${key}` };
      if (!tripId) {
        const res = await f(`${base}/v1/users?limit=1`, { headers, signal });
        return res.ok ? { ok: true, detail: "reachable; pick a trip to check its computer" } : { ok: false, detail: `HTTP ${res.status}` };
      }
      // Warming is the check: 200 means the computer is up, 202 that it is waking. A warm spends no credits.
      const res = await f(`${base}/v1/users/${v1UserIdFor(tripId)}/warm`, { method: 'POST', headers, signal });

      if (res.status === 202) return { ok: true, degraded: true, detail: 'computer waking, check again in a minute' };
      if (res.status === 404) return { ok: false, detail: 'this trip has no computer yet' };
      return { ok: false, detail: `HTTP ${res.status}` };
    },

type Query = { token?: string; trip?: string; note?: string };
type Req = FastifyRequest<{ Querystring: Query; Body: unknown }>;

function tokenOf(req: Req): string | undefined {
  const auth = req.headers.authorization;
  if (typeof auth === 'string' && auth.startsWith('Bearer ')) return auth.slice('Bearer '.length);
  return typeof req.query.token === 'string' ? req.query.token : undefined;
}

function sameToken(given: string | undefined, expected: string): boolean {
  if (!given) return false;
  return timingSafeEqual(createHash('sha256').update(given).digest(), createHash('sha256').update(expected).digest());
}

/** The body as an object, with empty form fields left out (an empty field means "not given"). */
function filled(body: unknown): Record<string, unknown> {
  if (!body || typeof body !== 'object') return {};
  return Object.fromEntries(Object.entries(body as Record<string, unknown>).filter(([, v]) => !(typeof v === 'string' && v.trim() === '')));
}

    };
    const switchValues = (): Array<{ name: SwitchName; value: string }> => SWITCH_NAMES.map((name) => ({ name, value: brain.switches.get(name) }));

      mode: brain.switches.get('CALLS_MODE'),

    });

        setTimeout(() => resolve(null), 1_000).unref();
      });
      const pack = await Promise.race([brain.packs.get(tripId).catch(() => null), timeout]);
      return pack ? pack.trip.people.map((p) => ({ id: p.person_id, name: p.name })) : [];

      const q = new URLSearchParams({ token: tokenOf(req) ?? '', ...(tripId ? { trip: tripId } : {}), note });
      return reply.code(303).header('location', `/admin?${q.toString()}`).send();
    };

    await brain.http.register(async (scope) => {
      if (!scope.hasContentTypeParser(FORM)) {
        scope.addContentTypeParser(FORM, { parseAs: 'string' }, (_req, body, done) => {
          done(null, Object.fromEntries(new URLSearchParams(String(body))));
        });
      }
      scope.addHook('onRequest', async (req, reply) => {
        if (!sameToken(tokenOf(req as Req), token)) {
          return reply.code(401).send(err('bad_signature', 'Wrong or missing admin token.'));
        }
      });

      scope.get<{ Querystring: Query }>('/admin', async (req, reply) => {
        const tripId = await tripOf(req.query);
        const [health, people, trips] = await Promise.all([checkAll(probesFor(brain, tripId)), peopleOf(tripId), newestTrips()]);
        const html = renderAdminPage({
          token, tripId, trips, at: brain.now(), health, switches: switchValues(), people, calls: calls(),
          note: req.query.note ?? null, docsUrl: DOCS_URL,
        });
        return reply.headers(HTML_HEADERS).type('text/html; charset=utf-8').send(html);
      });

      scope.get<{ Querystring: Query }>('/admin/health.json', async (req, reply) => {
        const tripId = await tripOf(req.query);
        const c = calls();
        const body: HealthJson = {
          at: brain.now().toISOString(),
          trip_id: tripId,
          server_socket_open: brain.door.link.isOpen(),
          probes: await checkAll(probesFor(brain, tripId)),
          switches: Object.fromEntries(switchValues().map((s) => [s.name, s.value])) as Record<SwitchName, string>,
          calls: { mode: c.mode, allowlist: c.allowlist.flatMap((a) => (a.name ? [a.name] : [])), allowlist_count: c.allowlist.length },
        };
        return reply.header('cache-control', 'no-store').send(body);
      });

      scope.post<{ Querystring: Query; Body: unknown }>('/admin/switches', async (req, reply) => {
        const body = (req.body && typeof req.body === 'object' ? req.body : {}) as Record<string, unknown>;
        const name = String(body['name'] ?? '');
        const value = String(body['value'] ?? '');
        const trip = req.query.trip ?? null;
        if (!Object.hasOwn(SWITCH_RULES, name)) return answer(req, reply, 422, err('invalid', `${name} is not a switch`), `${name} is not a switch`, trip);
        const sw = name as SwitchName;
        if (!SWITCH_RULES[sw].valid(value)) return answer(req, reply, 422, err('invalid', `${sw} cannot be "${value}"`), `${sw} cannot be "${value}"`, trip);
        await brain.switches.set(sw, value);
        brain.log.info({ name: sw, value }, 'admin: a switch changed');
        return answer(req, reply, 200, { ok: true, name: sw, value }, `${sw} is now ${value || 'none'}`, trip);
      });

      scope.post<{ Querystring: Query; Body: unknown }>('/admin/fire-watch', async (req, reply) => {
        const trip = req.query.trip ?? null;
        const parsed = FireForm.safeParse(filled(req.body));
        if (!parsed.success) return answer(req, reply, 422, err('invalid', z.prettifyError(parsed.error).slice(0, 300)), 'Pick the trip and the person first.', trip);
        const f = parsed.data;
        // P6's route, in process, with the demo's body: the same bar as any watch, and its card says "Simulated for the demo".
        const res = await brain.http.inject({
          method: 'POST',
          url: '/admin/watch/fire',
          headers: { authorization: `Bearer ${token}`, 'content-type': 'application/json' },
          payload: {
            trip_id: f.trip_id, label: f.label, person_ids: [f.person_id], delay_minutes: f.delay_minutes,
            ...(f.note ? { note: f.note } : {}), importance: 1, urgency: 1, reset_budget: true, force: true,
          },
        });
        let out: unknown;
        try {
          out = res.json();
        } catch {
          out = err('unavailable', res.body.slice(0, 200));
        }
        const promoted = (out as { promoted?: unknown } | null)?.promoted;
        const note = res.statusCode !== 200
          ? `The watch did not fire (HTTP ${res.statusCode}).`
          : promoted === 1 ? 'The watch fired: one nudge is on its way.' : 'The watch fired, but the bar held it back.';
        return answer(req, reply, res.statusCode, out, note, f.trip_id);
      });

      scope.post<{ Querystring: Query; Body: unknown }>('/admin/stop-jobs', async (req, reply) => {
        const parsed = TripForm.safeParse(filled(req.body));
        if (!parsed.success) return answer(req, reply, 422, err('invalid', 'trip_id is required'), 'Pick a trip first.', req.query.trip ?? null);
        const tripId = parsed.data.trip_id;
        const tasks = brain.tasks;
        if (!tasks) return answer(req, reply, 409, err('conflict', 'Jobs are off on this Go.'), 'Jobs are off on this Go.', tripId);

        for (const t of running) await tasks.stop(t.id);
        const ids = running.map((t) => t.id);
        const note = ids.length === 0 ? 'No job was running.' : `Stopped ${ids.length} running job${ids.length === 1 ? '' : 's'}.`;
        return answer(req, reply, 200, { ok: true, stopped: ids }, note, tripId);
      });

      scope.post<{ Querystring: Query; Body: unknown }>('/admin/decline-pay', async (req, reply) => {
        const parsed = TripForm.safeParse(filled(req.body));
        if (!parsed.success) return answer(req, reply, 422, err('invalid', 'trip_id is required'), 'Pick a trip first.', req.query.trip ?? null);
        const tripId = parsed.data.trip_id;
        const { rows } = await brain.pool.query<{ card: C.Card }>(
          `select card from cards where trip_id = $1 and kind = 'pay' and card->>'state' = 'waiting' order by updated_at desc limit 1`,
          [tripId],
        );
        const card = rows[0]?.card;
        if (!card || card.kind !== 'pay') return answer(req, reply, 404, err('not_found', 'No pay card is waiting in this trip.'), 'No pay card is waiting.', tripId);
        // The payer's own Decline, through P1's tap router: one answer per tap, and P5 releases the Server's approval.

        });
        const ok = result.result !== 'refused';
        const why = 'toast' in result && result.toast ? result.toast : 'refused';
        return answer(req, reply, ok ? 200 : 409, { ok, card_id: card.card_id, answer: result }, ok ? 'The waiting pay card was declined.' : `Not declined: ${why}`, tripId);
      });
    });
  };
}

const silent = (reason: string): Reply => ({ tools: [{ name: 'stay_silent', input: { reason } }] });
const blocks = (content: string | Block[] | undefined): Block[] => (Array.isArray(content) ? content : []);

/** The results of the round that just ran, each with the name of the tool that made it. */
export function lastResults(body: MessageBody): Array<{ name: string; content: string }> {
  const last = body.messages[body.messages.length - 1];
  const prev = body.messages[body.messages.length - 2];
  const names = new Map(blocks(prev?.content).filter((b) => b['type'] === 'tool_use').map((b) => [String(b['id']), String(b['name'])]));
  return blocks(last?.content)
    .filter((b) => b['type'] === 'tool_result')
    .map((b) => ({ name: names.get(String(b['tool_use_id'])) ?? '', content: typeof b['content'] === 'string' ? b['content'] : JSON.stringify(b['content']) }));
}

function lastAssistantText(body: MessageBody): string {
  for (let k = body.messages.length - 1; k >= 0; k--) {
    const m = body.messages[k];
    if (m?.role !== 'assistant') continue;
    if (typeof m.content === 'string') return m.content;
    return blocks(m.content).filter((b) => b['type'] === 'text').map((b) => String(b['text'] ?? '')).join('');
  }
  return '';
}

/** A string field of a tool result's JSON, or null. */
function field(content: string, key: string): string | null {
  try {
    const v = JSON.parse(content) as Record<string, unknown> | null;
    const x = v?.[key];
    return typeof x === 'string' ? x : null;
  } catch {
    return null;
  }
}

function messageOf(id: string, model: string, r: Reply, n: number) {

}

  function afterRound(body: MessageBody, ran: Array<{ name: string; content: string }>): Reply {

    const result = (name: string) => ran.find((r) => r.name === name);
    const searched = result('search_flights');
    if (searched) {
      const id = field(searched.content, 'results_id');
      return id ? { tools: [{ name: 'show_options', input: { results_id: id, title: 'Friday flights, Delhi to Goa' } }] } : { text: DIRECTOR_WORDS.noFlights };
    }

      return { text: DIRECTOR_WORDS.flights };

    if (result('put_to_vote')) return { text: DIRECTOR_WORDS.villasShown };
    if (result('start_task')) return { text: DIRECTOR_WORDS.villaStarted };
    if (result('set_mandate')) return { text: DIRECTOR_WORDS.mandate };

    },
  };

/** Twilio's REST API as P6's client calls it. On a new call it opens Go's media socket, as Twilio does. */
export function fakeTwilio(): FakeTwilio {
  let base: string | null = null;
  const calls: FakeTwilio['calls'] = [];
  const hangUps: string[] = [];
  const sockets = new Map<string, WebSocket>();
  const json = (status: number, body: unknown): Response =>
    new Response(JSON.stringify(body), { status, headers: { 'content-type': 'application/json' } });

  const dialGo = (sid: string, path: string, token: string): void => {
    if (base === null) return;
    const ws = new WebSocket(`${base}${path}`);
    sockets.set(sid, ws);
    ws.on('open', () => {
      ws.send(JSON.stringify({ event: 'connected', protocol: 'Call', version: '1.0.0' }));
      ws.send(JSON.stringify({ event: 'start', streamSid: `MZ_${sid}`, start: { streamSid: `MZ_${sid}`, callSid: sid, customParameters: { token } } }));
      ws.send(JSON.stringify({ event: 'media', streamSid: `MZ_${sid}`, media: { timestamp: '0', payload: SILENCE } }));
    });
    ws.on('error', () => undefined);
  };

  const fetchImpl = (async (input: string | URL | Request, init?: RequestInit) => {
    const url = String(input instanceof Request ? input.url : input);
    const form = new URLSearchParams(String(init?.body ?? ''));
    const hangUp = /\/Calls\/([A-Za-z0-9_]+)\.json$/.exec(url);
    if (hangUp?.[1]) {
      const sid = hangUp[1];
      hangUps.push(sid);
      sockets.get(sid)?.close();
      sockets.delete(sid);
      return json(200, { sid, status: 'completed' });
    }
    if (url.endsWith('/Calls.json')) {
      const twiml = form.get('Twiml') ?? '';
      const streamUrl = /<Stream url="([^"]+)"/.exec(twiml)?.[1] ?? '';
      const token = /name="token" value="([^"]+)"/.exec(twiml)?.[1] ?? '';
      const sid = `CA_demo_${calls.length + 1}`;
      const path = streamUrl ? new URL(streamUrl.replace(/&amp;/g, '&')).pathname : '';
      calls.push({ sid, to: form.get('To') ?? '', from: form.get('From') ?? '', path });
      setTimeout(() => dialGo(sid, path, token), 50);
      return json(201, { sid, status: 'queued' });
    }
    return json(404, { message: `the fake Twilio has no ${url}` });
  }) as typeof fetch;

  return {
    fetch: fetchImpl,
    calls,
    hangUps,
    connectTo(goWsBase) {
      base = goWsBase;
    },
    closeAll() {
      for (const ws of sockets.values()) ws.close();
      sockets.clear();
    },
  };
}

/**
 * The rehearsal's beat timer and its sheet: each beat of the run sheet with its budget, how long it took and PASS or
 * FAIL, printed as the run sheet's table; and the checks that stop a rehearsal before it starts.
 */
import type { HealthJson } from '@go/brain/dist/admin/register.js';
import { beatOf, DEMO_BEATS, type Beat, type BeatId } from '../scenarios/demo.js';

export type BeatResult = 'PASS' | 'FAIL' | 'NOT REACHED';

export interface BeatRun extends Beat {
  /** From the cue to what had to happen; null when it never happened or is not timed. */
  ms: number | null;
  result: BeatResult;
  note: string;
}

export const secs = (ms: number): string => `${(ms / 1000).toFixed(1)} s`;

export class BeatTimer {
  private readonly runs = new Map<BeatId, BeatRun>();

  constructor(private readonly clock: () => number = () => Date.now()) {}

  /** The beat happened now, `startedAt` after its cue. It passes inside its budget, unless `ok` says otherwise. */
  done(id: BeatId, startedAt: number, o: { ok?: boolean; note?: string } = {}): BeatRun {
    const beat = beatOf(id);
    const ms = Math.max(0, this.clock() - startedAt);
    const budget = beat.budgetMs;
    const inBudget = budget === null || ms <= budget;
    const note = o.note ?? (budget === null || inBudget ? '' : `over by ${secs(ms - budget)}`);
    return this.keep({ ...beat, ms, result: (o.ok ?? true) && inBudget ? 'PASS' : 'FAIL', note });
  }

  /** A beat that is checked, not timed (quiet through the chatter). */
  check(id: BeatId, ok: boolean, note: string): BeatRun {
    return this.keep({ ...beatOf(id), ms: null, result: ok ? 'PASS' : 'FAIL', note });
  }

  /** The beat never happened. */
  miss(id: BeatId, note: string): BeatRun {
    return this.keep({ ...beatOf(id), ms: null, result: 'FAIL', note });
  }

  /** Every beat in the run sheet's order; a beat with no result was not reached. */
  results(): BeatRun[] {
    return DEMO_BEATS.map((b) => this.runs.get(b.id) ?? { ...b, ms: null, result: 'NOT REACHED', note: '' });
  }

  private keep(run: BeatRun): BeatRun {
    const first = this.runs.get(run.id);
    if (first) return first;
    this.runs.set(run.id, run);
    return run;
  }
}

export function passed(runs: BeatRun[]): boolean {
  return runs.every((r) => r.result === 'PASS');
}

/** The timed beat closest to (or furthest over) its budget. */
function slowest(runs: BeatRun[]): string {
  let worst: { run: BeatRun; ms: number; budget: number } | null = null;
  for (const run of runs) {
    if (run.ms === null || run.budgetMs === null) continue;
    if (worst === null || run.ms / run.budgetMs > worst.ms / worst.budget) worst = { run, ms: run.ms, budget: run.budgetMs };
  }
  return worst ? `${worst.run.name}, ${secs(worst.ms)} of ${secs(worst.budget)}` : 'none timed';
}

    '',

}

/** Waits for a beat and records it: PASS inside its budget and when `check` finds nothing wrong. */
async function timed<T>(timer: BeatTimer, id: BeatId, startedAt: number, wait: (ms: number) => Promise<T>, check?: (v: T) => string | null): Promise<T | null> {
  const ms = patienceMs(id);
  try {
    const v = await wait(ms);
    const problem = check ? check(v) : null;
    timer.done(id, startedAt, problem === null ? {} : { ok: false, note: problem });
    return v;
  } catch {
    timer.miss(id, `nothing in ${secs(ms)}`);
    return null;
  }
}

export async function rehearse(o: RehearseOptions): Promise<RehearseOutcome> {
  const log = o.log ?? ((line: string) => console.log(line));
  const now = o.now ?? (() => new Date());
  if (o.warm) return warmStage(o, log);
  if (!o.secret) throw new RefusedError(['A full rehearsal connects as the Server: give --secret (or GO_SIGNING_SECRET).']);
  const health = await readHealth(o);
  for (const p of health.probes) log(`${p.dot.toUpperCase().padEnd(5)} ${p.label}: ${p.detail} (${p.ms} ms)`);
  const reasons = preflight(health);
  if (reasons.length > 0) throw new RefusedError(reasons);

  const trip = rehearsalTrip(now());
  const fake = new FakeServer({ goUrl: o.go, secret: o.secret, instance: `rehearsal-${trip.trip_id.slice('t_reh_'.length)}` });
  await fake.start(15_000);
  const room = new RehearsalRoom(fake, trip.trip_id);
  const timer = new BeatTimer();
  try {
    await fake.createTrip(trip);
    await room.waitGo(0, (m) => !!m.text, 30_000);
    log(`Rehearsal trip ${trip.trip_id}: Go said hello.`);
    await playBeats(room, timer, o, log);
    return { tripId: trip.trip_id, runs: timer.results(), warmed: false, health };
  } finally {
    await adminPost(o, '/admin/stop-jobs', { trip_id: trip.trip_id }).catch(() => null);
    fake.stop();
  }
}

const bare = (url: string): string => url.replace(/\/+$/, '');

/**
 * Flags first, then the environment. A full run goes to GO_REHEARSAL_URL and never to GO_STAGE_URL; --warm goes to
 * GO_STAGE_URL and needs no secret. `--new-trip` is accepted: every run makes a fresh trip (interface note 6).
 */
export function parseRehearseArgs(argv: string[], env: Record<string, string | undefined>): RehearseOptions {
  const { values } = parseArgs({
    args: argv.filter((a) => a !== '--'),
    options: {
      go: { type: 'string' },
      secret: { type: 'string' },
      'admin-token': { type: 'string' },
      warm: { type: 'boolean' },
      'new-trip': { type: 'boolean' },
      trip: { type: 'string' },
    },
  });
  const warm = values.warm === true;
  const raw = values.go ?? (warm ? env['GO_STAGE_URL'] : env['GO_REHEARSAL_URL']);
  if (!raw) {
    throw new Error(warm
      ? 'Give --go <url> (or GO_STAGE_URL): the stage Go whose health read --warm makes.'
      : 'Give --go <url> (or GO_REHEARSAL_URL): brain-rehearsal, the Go a full rehearsal runs on.');
  }
  const go = bare(raw);
  const stage = env['GO_STAGE_URL'];
  if (!warm && stage && bare(stage) === go) {
    throw new Error('That is the stage Go (GO_STAGE_URL). Rehearsals never share it: give --go brain-rehearsal\'s URL, or use --warm for the health read.');
  }
  const secret = values.secret ?? env['GO_SIGNING_SECRET'];
  if (!warm && !secret) throw new Error('Give --secret <GO_SIGNING_SECRET> (or set GO_SIGNING_SECRET): a full run connects as the Server.');
  const adminToken = values['admin-token'] ?? env['GO_ADMIN_TOKEN'];
  if (!adminToken) throw new Error('Give --admin-token <ADMIN_TOKEN> (or set GO_ADMIN_TOKEN): the runner reads /admin/health.json first.');
  return {
    go,
    ...(!warm && secret ? { secret } : {}),
    adminToken,
    warm,
    ...(values.trip ? { trip: values.trip } : {}),
  };
}

  if (text === null) return { bySeq, seqOfLine };
  text.split('\n').forEach((raw, n) => {
    const row = raw.trim();
    if (row === '') return;
    let v: unknown;
    try {
      v = JSON.parse(row);
    } catch {
      throw new Error(`the existing file's line ${n + 1} is not JSON`);
    }
    if (!v || typeof v !== 'object') return;
    const o = v as Record<string, unknown>;
    const seq = o['seq'];
    const i = o['i'];
    if (typeof seq === 'number' && typeof i === 'number') {
      bySeq.set(seq, o);
      seqOfLine.set(i, seq);
    }
  });
  return { bySeq, seqOfLine };
}

export async function exportTrip(pool: pg.Pool, tripId: string, existing: string | null): Promise<TripExport> {
  const prior = readPrior(existing);
  const { rows } = await pool.query<Row>(
    'select message_id, seq, person_id, text, reply_to, deleted_at, payload from messages where trip_id = $1 order by seq',
    [tripId],
  );
  const out: Array<Record<string, unknown>> = [];
  const lineOfMessage = new Map<string, number>();
  const lineOfSeq = new Map<number, number>();
  const droppedProbes: string[] = [];
  let kept = 0;
  let added = 0;
  let skipped = 0;
  let start: number | null = null;
  let lastAt = 0;

  for (const r of rows) {
    const m = r.payload;
    const isGo = m.author.kind === 'go';
    const personId = m.author.kind === 'person' ? (r.person_id ?? m.author.person_id) : null;
    if (r.deleted_at !== null || m.deleted || m.withheld || (!isGo && personId === null)) {
      skipped += 1;
      continue;
    }
    const caption = m.attachments.find((a): a is Photo => a.kind === 'photo')?.caption ?? null;
    const raw = isGo ? (r.text ?? m.card?.fallback_text ?? null) : (r.text ?? caption);
    const text = raw === null ? '' : clip(screen(raw), 4000);
    if (text === '') {
      skipped += 1;
      continue;
    }
    const sentMs = Date.parse(m.sent_at);
    start ??= sentMs;
    const at = Math.max(lastAt, Math.round((sentMs - start) / 100) / 10);
    lastAt = at;

    const i = out.length;
    const line: Record<string, unknown> = { i, at, kind: 'say', person: isGo ? 'go' : benchPersonId(personId ?? ''), text, seq: r.seq };
    if (!isGo && caption) {
      const shown = clip(screen(caption), 500);
      if (shown !== '') line['photo'] = { caption: shown };
    }
    const replyLine = r.reply_to === null ? undefined : lineOfMessage.get(r.reply_to);
    if (replyLine !== undefined) line['reply_to'] = replyLine;

    if (isGo) {
      line['go_should_speak'] = 'either';
    } else {
      const p = prior.bySeq.get(r.seq);
      if (p) {
        kept += 1;
        for (const k of LABELS) if (p[k] !== undefined) line[k] = p[k];
        line['go_should_speak'] ??= addressesGo(text) ? true : 'either';
        const probes = Array.isArray(p['recall']) ? (p['recall'] as Probe[]) : [];
        const moved = probes.flatMap((probe) => {
          const proofSeq = prior.seqOfLine.get(probe.line);
          const to = proofSeq === undefined ? undefined : lineOfSeq.get(proofSeq);
          if (to === undefined) {
            droppedProbes.push(`seq ${r.seq}: "${probe.question}" (its proof message is gone)`);
            return [];
          }
          return [{ ...probe, line: to }];
        });
        if (moved.length > 0) line['recall'] = moved;
      } else {
        added += 1;
        line['go_should_speak'] = addressesGo(text) ? true : 'either';
        line['note'] = NEW_LINE_NOTE;
      }
    }
    out.push(line);
    lineOfMessage.set(r.message_id, i);
    lineOfSeq.set(r.seq, i);
  }
  return { lines: out.map((l) => JSON.stringify(l)), kept, added, skipped, droppedProbes };
}
